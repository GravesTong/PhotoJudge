```yaml

id: COMP-001              # 维度前缀 + 序号（COMP/EXP/FOC/COL/SUB/PPC）
rule: 规则一句话           # 如"主体应避开正中心，或置于三分线交叉点"
why: 可解释理由            # 为什么这条成立（原理层面）

source:                   # 来源名 + URL
evidence:                 # 已查证 / 一方称 / 通识
conflicts:                # 来源间冲突（如有）
```
